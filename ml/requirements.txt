# ML-only dependencies for the Resume-to-Job Match Scorer.
# Kept separate from backend/requirements.txt so the core API never
# depends on the ML stack (spec: isolated package, CPU-only).
# Pinned exactly: the committed match_scorer.joblib artifact must unpickle
# against the same library versions it was trained with.
#
# --extra-index-url pulls the CPU-only torch build (+cpu local version) on
# Linux. Without it, pip resolves plain PyPI torch on Linux, which bundles
# the full CUDA runtime (nvidia-cublas-cu12 etc.) — several GB that blow
# past Render's free-tier build limits. The +cpu wheel has no CUDA deps.
# macOS has no +cpu variant (no CUDA split to begin with), so dev machines
# still resolve the plain wheel via the marker below.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.13.0+cpu ; sys_platform == "linux"
torch==2.13.0 ; sys_platform != "linux"
numpy==2.5.1
pandas==3.0.3
scikit-learn==1.9.0
xgboost==3.3.0
sentence-transformers==5.6.0
joblib==1.5.3
